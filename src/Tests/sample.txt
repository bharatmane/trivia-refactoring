    <script>
        function openTab(tabName) {
            const tabContents = document.querySelectorAll('.tab-content');
            tabContents.forEach(content => content.classList.remove('active'));
            const tabButtons = document.querySelectorAll('.tab-btn');
            tabButtons.forEach(btn => btn.classList.remove('active'));
            document.getElementById(tabName).classList.add('active');
            event.target.classList.add('active');
        }

        const modal = document.getElementById('policy-modal');
        const addBtn = document.getElementById('add-new-policy-btn');
        const closeBtn = document.getElementById('close-modal');
        const cancelBtn = document.getElementById('cancel-btn');
        const form = document.getElementById('policy-form');
        const tableBody = document.getElementById('policy-table-body');
        const modalTitle = document.getElementById('modal-title');
        const saveBtn = document.getElementById('save-btn');
        const unsavedIndicator = document.getElementById('unsaved-indicator');
        let editingRow = null;
        let unsavedChanges = false;

        addBtn.onclick = function() {
            modalTitle.textContent = 'Add New Policy';
            form.reset();
            editingRow = null;
            modal.style.display = 'block';
        }

        closeBtn.onclick = function() {
            modal.style.display = 'none';
            unsavedChanges = true;
            unsavedIndicator.style.display = 'block';
        }

        cancelBtn.onclick = function() {
            modal.style.display = 'none';

        form.onsubmit = function(e) {
            e.preventDefault();
            const imSource = document.getElementById('im-source').value;
            const policy = document.getElementById('policy').value;
            const businessAreas = Array.from(document.querySelectorAll('input[name="business-area"]:checked')).map(cb => cb.value).join(', ');
            const confidenceLevel = document.getElementById('confidence-level').value + '%';
            const businessDays = document.getElementById('business-days').value;

            if (editingRow) {
                editingRow.cells[0].textContent = businessAreas;
                editingRow.cells[1].textContent = policy;
                editingRow.cells[2].textContent = imSource;
                editingRow.cells[3].textContent = confidenceLevel;
                editingRow.cells[4].textContent = businessDays;
            } else {
                const row = tableBody.insertRow();
                row.classList.add('unsaved-policy');
                row.innerHTML = `
                    <td>${businessAreas}</td>
                    <td>${policy}</td>
                    <td>${imSource}</td>
                    <td>${confidenceLevel}</td>
                    <td>${businessDays}</td>
                    <td>
                        <button class="edit-btn" onclick="editPolicy(this)">Edit</button>
                        <button class="delete-btn" onclick="deletePolicy(this)">Delete</button>
                    </td>
                `;
            }
            modal.style.display = 'none';
        }

        function editPolicy(btn) {
            editingRow = btn.parentElement.parentElement;
            editingRow.classList.add('unsaved-policy');
            modalTitle.textContent = 'Edit Policy';
            const cells = editingRow.cells;
            document.getElementById('im-source').value = cells[2].textContent;
            document.getElementById('policy').value = cells[1].textContent;
            const areas = cells[0].textContent.split(', ');
            document.querySelectorAll('input[name="business-area"]').forEach(cb => {
                cb.checked = areas.includes(cb.value);
            });
            document.getElementById('confidence-level').value = parseInt(cells[3].textContent);
            document.getElementById('business-days').value = cells[4].textContent;
            modal.style.display = 'block';
        }

            if (confirm('Are you sure you want to delete this policy?')) {
                btn.parentElement.parentElement.remove();
                unsavedChanges = true;
                unsavedIndicator.style.display = 'block';
            }
        }

        // IA Config functions
        const iaModal = document.getElementById('addPolicyModal');
        const iaForm = document.getElementById('addPolicyForm');
        const iaTableBody = document.getElementById('ia-table-body');
        const iaUnsavedIndicator = document.getElementById('ia-unsaved-indicator');
        let iaEditingRow = null;

        function openAddPolicyModal() {
            document.getElementById('modalTitle').textContent = 'Add New Policy';
            iaForm.reset();
            iaEditingRow = null;
            iaModal.style.display = 'block';
        }

        function openEditPolicyModal(btn) {
            iaEditingRow = btn.closest('tr');
            const cells = iaEditingRow.cells;
            document.getElementById('modalTitle').textContent = 'Edit Policy';
            document.getElementById('category').value = cells[0].textContent;
            document.getElementById('policy').value = cells[1].textContent;
            document.getElementById('direction').value = cells[2].textContent;
            document.getElementById('livePolicy').checked = cells[3].querySelector('input').checked;
            document.getElementById('isTmPolicy').checked = cells[4].querySelector('input').checked;
            const detailsRow = iaEditingRow.nextElementSibling;
            if (detailsRow && detailsRow.classList.contains('details-row')) {
                const paramText = detailsRow.cells[0].textContent;
                const param = paramText.split(' - ')[1];
                document.getElementById('parameters').value = param;
            }
            iaModal.style.display = 'block';
        }

        function closeModal() {
            iaModal.style.display = 'none';
        }

        function toggleDetails(btn) {
            const row = btn.closest('tr');
            const detailsRow = row.nextElementSibling;
            if (detailsRow && detailsRow.classList.contains('details-row')) {
                detailsRow.style.display = detailsRow.style.display === 'none' ? 'table-row' : 'none';
                btn.textContent = btn.textContent === '+' ? '-' : '+';
            }
        }

        iaForm.onsubmit = function(e) {
            e.preventDefault();
            const category = document.getElementById('category').value;
            const policy = document.getElementById('policy').value;
            const direction = document.getElementById('direction').value;
            const livePolicy = document.getElementById('livePolicy').checked ? 'checked' : '';
            const isTmPolicy = document.getElementById('isTmPolicy').checked ? 'checked' : '';
            const parameters = document.getElementById('parameters').value;

            if (iaEditingRow) {
                // Update existing row
                const cells = iaEditingRow.cells;
                cells[0].textContent = category;
                cells[1].textContent = policy;
                cells[2].textContent = direction;
                cells[3].innerHTML = `<input type="checkbox" ${livePolicy} />`;
                cells[4].innerHTML = `<input type="checkbox" ${isTmPolicy} />`;
                // Update details
                const detailsRow = iaEditingRow.nextElementSibling;
                if (detailsRow && detailsRow.classList.contains('details-row')) {
                    detailsRow.cells[0].textContent = `Details: Parameter - ${parameters}`;
                }
                iaEditingRow.classList.add('unsaved-policy');
            } else {
                // Add new row
                const row = iaTableBody.insertRow();
                row.classList.add('unsaved-policy');
                row.innerHTML = `
                    <td>${category}</td>
                    <td>${policy}</td>
                    <td>${direction}</td>
                    <td><input type="checkbox" ${livePolicy} /></td>
                    <td><input type="checkbox" ${isTmPolicy} /></td>
                    <td>
                        <button class="edit-btn" onclick="openEditPolicyModal(this)">Edit</button>
                        <button class="delete-btn">Delete</button>
                    </td>
                    <td><button onclick="toggleDetails(this)">+</button></td>
                `;
                const detailsRow = iaTableBody.insertRow();
                detailsRow.classList.add('details-row');
                detailsRow.style.display = 'none';
                detailsRow.innerHTML = `<td colspan="7">Details: Parameter - ${parameters}</td>`;
            }

            iaUnsavedIndicator.style.display = 'block';
            iaModal.style.display = 'none';
        }
